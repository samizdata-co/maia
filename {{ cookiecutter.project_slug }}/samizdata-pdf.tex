% PDF delivery overrides: LuaLaTeX cannot obtain Bunny-hosted fonts, so use
% installed Space Grotesk and Work Sans, or the vendored copies in fonts/.
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{titlesec}
\usepackage[automark]{scrlayer-scrpage}

% Documents may live in the project root or one folder down (e.g. docs/).
% The path is absolute because luaotfload caches fonts by name along with the
% first path it saw; a relative path breaks documents in the other folder.
\IfFileExists{fonts/WorkSans-Variable.ttf}
  {\edef\samizdatafontpath{\directlua{tex.sprint(lfs.currentdir())}/fonts/}}
  {\edef\samizdatafontpath{\directlua{tex.sprint(lfs.currentdir())}/../fonts/}}

% Prefer installed fonts: that cache also outlives a deleted or moved project,
% so the vendored copies are only a fallback.
\IfFontExistsTF{Work Sans}
  {\setmainfont{Work Sans}\setsansfont{Work Sans}}
  {
    % The vendored fonts are variable, so pin each weight explicitly;
    % otherwise LuaLaTeX uses the default instance and has no bold.
    \defaultfontfeatures[WorkSans]{
      Path=\samizdatafontpath,
      UprightFont=WorkSans-Variable.ttf,
      UprightFeatures={RawFeature={axis={wght=400}}},
      BoldFont=WorkSans-Variable.ttf,
      BoldFeatures={RawFeature={axis={wght=700}}},
      ItalicFont=WorkSans-Italic-Variable.ttf,
      ItalicFeatures={RawFeature={axis={wght=400}}},
      BoldItalicFont=WorkSans-Italic-Variable.ttf,
      BoldItalicFeatures={RawFeature={axis={wght=700}}}
    }%
    \setmainfont{WorkSans}\setsansfont{WorkSans}%
  }
\IfFontExistsTF{Space Grotesk}
  {\newfontfamily\samizdatadisplay{Space Grotesk}}
  {
    \newfontfamily\samizdatadisplay{Space Grotesk}[
      Path=\samizdatafontpath,
      UprightFont=SpaceGrotesk-Variable.ttf,
      UprightFeatures={RawFeature={axis={wght=400}}},
      BoldFont=SpaceGrotesk-Variable.ttf,
      BoldFeatures={RawFeature={axis={wght=700}}}
    ]%
  }
